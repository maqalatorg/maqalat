## 1) نسخ احتياطي تلقائي

```python
import shutil
from datetime import datetime
from pathlib import Path

def backup_folder(source: str, backup_root: str) -> str:
    """نسخة احتياطية بالتاريخ."""
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M")
    backup_dir = Path(backup_root) / f"backup_{timestamp}"
    shutil.copytree(source, backup_dir)
    return str(backup_dir)

# استخدام
backup_folder("/Users/you/Documents/Important", "/Users/you/Backups")
```

## 3) قراءة إيميلات (IMAP)

```python
import imaplib
import email
from email.header import decode_header

def read_unread_emails(user: str, password: str, imap_server: str = "imap.gmail.com"):
    """قراءة الإيميلات غير المقروءة."""
    with imaplib.IMAP4_SSL(imap_server) as mail:
        mail.login(user, password)
        mail.select("inbox")
        
        _, msg_ids = mail.search(None, "UNSEEN")
        
        for msg_id in msg_ids[0].split():
            _, msg_data = mail.fetch(msg_id, "(RFC822)")
            email_body = email.message_from_bytes(msg_data[0][1])
            
            subject = decode_header(email_body["Subject"])[0][0]
            if isinstance(subject, bytes):
                subject = subject.decode()
            
            print(f"من: {email_body['From']}")
            print(f"الموضوع: {subject}\n")
```

## 4) تصفية الصور بالحجم

```python
from PIL import Image
from pathlib import Path

def compress_images(folder: str, max_size_kb: int = 500) -> None:
    """ضغط الصور الكبيرة."""
    for img_path in Path(folder).glob("*.jpg"):
        if img_path.stat().st_size > max_size_kb * 1024:
            img = Image.open(img_path)
            img.thumbnail((1920, 1920))
            img.save(img_path, quality=80, optimize=True)
            print(f"ضُغطت: {img_path.name}")
```

## 5) تقرير أسبوعي تلقائي

```python
import pandas as pd
from datetime import datetime, timedelta

def weekly_report(csv_file: str) -> str:
    """تقرير الأسبوع الماضي من CSV."""
    df = pd.read_csv(csv_file, parse_dates=["date"])
    
    week_ago = datetime.now() - timedelta(days=7)
    recent = df[df["date"] >= week_ago]
    
    report = f"""
    # تقرير الأسبوع {datetime.now().strftime('%Y-%m-%d')}
    
    - إجمالي المعاملات: {len(recent)}
    - المبيعات: {recent['amount'].sum():,.2f} ريال
    - المتوسّط: {recent['amount'].mean():,.2f} ريال
    - أعلى معاملة: {recent['amount'].max():,.2f} ريال
    """
    return report
```

## 6) مراقبة موقع (Ping)

```python
import requests
import time

def monitor_website(url: str, check_interval: int = 60) -> None:
    """مراقبة موقع وإرسال تنبيه إذا سقط."""
    while True:
        try:
            r = requests.get(url, timeout=10)
            if r.status_code != 200:
                print(f"⚠️ {url} — HTTP {r.status_code}")
                # send_alert()
        except Exception as e:
            print(f"❌ {url} — {e}")
            # send_alert()
        
        time.sleep(check_interval)
```

## 7) تحويل PDF إلى نصّ

```python
import pypdf

def pdf_to_text(pdf_path: str) -> str:
    """استخراج نصّ من PDF."""
    reader = pypdf.PdfReader(pdf_path)
    text = ""
    for page in reader.pages:
        text += page.extract_text() + "\n"
    return text

# استخدام
```

## 8) تعديل جماعي لأسماء الملفّات

```python
from pathlib import Path

def bulk_rename(folder: str, pattern: str, replacement: str) -> None:
    """تعديل جماعي لأسماء الملفّات."""
    for file in Path(folder).iterdir():
        if pattern in file.name:
            new_name = file.name.replace(pattern, replacement)
            file.rename(file.parent / new_name)
            print(f"{file.name} → {new_name}")

# استخدام
bulk_rename("/photos/vacation", "IMG_", "vacation_")
```

## 9) استخراج بيانات من Excel لـJSON

```python
import pandas as pd
import json

def excel_to_json(excel_path: str, json_path: str, sheet: str = None) -> None:
    """تحويل Excel إلى JSON."""
    df = pd.read_excel(excel_path, sheet_name=sheet)
    data = df.to_dict(orient="records")
    
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

excel_to_json("customers.xlsx", "customers.json")
```
